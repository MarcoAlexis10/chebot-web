create or replace function concierge_service_qa_v1.auth_live(p_user uuid,p_session uuid,p_exp bigint) returns void
language plpgsql security definer set search_path='' set lock_timeout='2s' as $$begin
 if current_user<>'cheveni_service_auth_owner_v2' or session_user not in('cheveni_service_manager_backend_v2','cheveni_service_reader_backend_v2','cheveni_service_ops_backend_v2') or p_user is null or p_session is null or p_exp is null or
 p_exp<=extract(epoch from clock_timestamp()) or not exists(select from auth.users u join auth.sessions s on s.user_id=u.id
 where u.id=p_user and s.id=p_session and u.email_confirmed_at is not null and u.email_confirmed_at<=clock_timestamp()
 and (u.banned_until is null or u.banned_until<=clock_timestamp()) and(s.not_after is null or s.not_after>clock_timestamp()))
 then raise insufficient_privilege using message='SERVICE_SESSION_DENIED';end if;
end$$;
