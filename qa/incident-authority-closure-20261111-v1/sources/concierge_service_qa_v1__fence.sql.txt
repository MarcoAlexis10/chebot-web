create or replace function concierge_service_qa_v1.fence() returns void language plpgsql security invoker set search_path='' as $$
declare t text;n integer;begin
 if not ((session_user::text='cheveni_service_manager_backend_v2' and current_user::text='cheveni_service_write_owner_v2') or (session_user::text='cheveni_service_reader_backend_v2' and current_user::text='cheveni_service_read_owner_v2') or (session_user::text='cheveni_service_ops_backend_v2' and current_user::text='cheveni_service_ops_owner_v2')) then raise insufficient_privilege using message='SERVICE_CALLER_PAIR_HELD';end if;

 if current_setting('session_replication_role')<>'origin' or current_setting('transaction_isolation')<>'read committed' then raise insufficient_privilege using message='SERVICE_ISOLATION_DENIED';end if;
 if not exists(select from pg_proc where oid=to_regprocedure('concierge_private.manager_fence_lock_v1()') and not prosecdef
 and proowner='postgres'::regrole and 'search_path=""'=any(proconfig)
 and not exists(select from aclexplode(coalesce(proacl,acldefault('f',proowner))) a where a.grantee<>proowner)
 and encode(sha256(convert_to(prosrc,'UTF8')),'hex')='10ea154cd7d63f91856346d0d9619e4796637c337561b92c64b29e0863acadc6')
 then raise insufficient_privilege using message='SERVICE_FENCE_DENIED';end if;
 foreach t in array array['auth.users','auth.sessions','public.concierge_accounts','public.concierge_properties',
 'public.concierge_account_members','public.concierge_manager_web_identities','public.concierge_member_property_permissions','public.concierge_stays'] loop
  select count(*) into n from pg_trigger where tgrelid=(select c.oid from pg_class c join pg_namespace ns on ns.oid=c.relnamespace where ns.nspname=split_part(t,'.',1) and c.relname=split_part(t,'.',2)) and tgname='cheveni_manager_fence_v1'
  and tgfoid='concierge_private.manager_fence_lock_v1()'::regprocedure and tgenabled in('O','A') and tgtype=62
  and not tgisinternal and tgqual is null and tgattr::text='' and tgnargs=0 and octet_length(tgargs)=0 and not tgdeferrable and not tginitdeferred and tgparentid=0 and tgconstraint=0;
  if n<>1 then raise insufficient_privilege using message='SERVICE_FENCE_COVERAGE';end if;
 end loop;
 foreach t in array array['concierge_service_qa_v1.launch_control','concierge_service_qa_v1.guest_bindings',
 'concierge_service_qa_v1.postcheckout_cases','concierge_service_qa_v1.ops_bindings','concierge_service_qa_v1.commands',
 'public.concierge_guest_identities','public.concierge_stay_members','public.concierge_guest_invites',
 'public.concierge_guest_phone_changes','public.concierge_incidents','public.concierge_operational_people','public.concierge_operational_tasks'] loop
  select count(*) into n from pg_trigger where tgrelid=(select c.oid from pg_class c join pg_namespace ns on ns.oid=c.relnamespace where ns.nspname=split_part(t,'.',1) and c.relname=split_part(t,'.',2)) and tgname='cheveni_service_fence_v1'
  and tgfoid='concierge_private.manager_fence_lock_v1()'::regprocedure and tgenabled in('O','A') and tgtype=62
  and not tgisinternal and tgqual is null and tgattr::text='' and tgnargs=0 and octet_length(tgargs)=0 and not tgdeferrable and not tginitdeferred and tgparentid=0 and tgconstraint=0;
  if n<>1 then raise insufficient_privilege using message='SERVICE_FENCE_COVERAGE';end if;
 end loop;
 perform pg_advisory_xact_lock(1129728581,1);
end$$;
