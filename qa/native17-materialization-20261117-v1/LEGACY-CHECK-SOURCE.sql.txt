create function concierge_invite_private.check_private_mutation_authority_v1(
 p_user uuid,p_session uuid,p_expiry bigint,p_member uuid,p_role text,p_revision bigint,
 p_account uuid,p_property uuid,p_operation text
) returns boolean language plpgsql stable security invoker set search_path='' as $$
declare c record;n bigint;begin
 if p_user is null or p_session is null or p_member is null or p_account is null
 or p_expiry is null or p_expiry<=extract(epoch from pg_catalog.clock_timestamp())
 or p_revision is null or p_revision<1 or p_role is null
 or p_role not in ('owner','manager','co_manager')
 or (p_operation='property_create' and p_role<>'owner') then return false;end if;
 select count(*) into n from concierge_invite_private.private_mutation_authority_v1(
  p_user,p_session,p_expiry,p_property,p_operation);
 if n<>1 then return false;end if;
 select * into c from concierge_invite_private.private_mutation_authority_v1(
  p_user,p_session,p_expiry,p_property,p_operation);
 return (c.account_id,c.member_id,c.member_role,c.authority_revision)
  is not distinct from (p_account,p_member,p_role,p_revision);
end$$;
