function concierge_service_qa_v1.launch(p_property uuid,p_require_review boolean default true) returns uuid
language plpgsql security invoker set search_path='' as $$declare a uuid;begin
 if not ((session_user::text='cheveni_service_manager_backend_v2' and current_user::text='cheveni_service_write_owner_v2') or (session_user::text='cheveni_service_reader_backend_v2' and current_user::text='cheveni_service_read_owner_v2') or (session_user::text='cheveni_service_ops_backend_v2' and current_user::text='cheveni_service_ops_owner_v2')) then raise insufficient_privilege using message='SERVICE_CALLER_PAIR_HELD';end if;

 select l.account_id into a from concierge_service_qa_v1.launch_control l join public.concierge_properties p
 on p.id=l.property_id and p.account_id=l.account_id join public.concierge_accounts x on x.id=l.account_id
 where l.singleton and l.property_id=p_property and(not p_require_review or l.state='REVIEW')
 and x.founding_manager and x.beta_started_at<=clock_timestamp() and x.beta_ends_at>clock_timestamp()
 and x.beta_ends_at<=x.beta_started_at+interval '30 days'
 and exists(select from public.concierge_manager_web_identities wi join public.concierge_account_members fm on fm.id=wi.member_id and fm.account_id=wi.account_id where wi.auth_user_id=l.founder_user_id and wi.account_id=l.account_id and wi.status='active' and wi.revoked_at is null and fm.role='owner' and fm.status='active')
 and p.status='active' and p.city in('Buenos Aires','CABA') and x.status in('active','testing')
 and l.charge_mode='NO_BILLING' and not l.guest_access_enabled and not l.codes_allowed
 and (select count(*) from public.concierge_properties z where z.account_id=l.account_id and z.status='active')=1;
 if a is null then raise insufficient_privilege using message='SERVICE_LAUNCH_HELD';end if;return a;
end$$;

function concierge_service_qa_v1.manager_scope(p_user uuid,p_session uuid,p_exp bigint,p_property uuid,p_action text,p_review boolean default true)
returns jsonb language plpgsql security invoker set search_path='' as $$declare ctx jsonb;a uuid;begin
 if not ((session_user::text='cheveni_service_manager_backend_v2' and current_user::text='cheveni_service_write_owner_v2') or (session_user::text='cheveni_service_reader_backend_v2' and current_user::text='cheveni_service_read_owner_v2')) then raise insufficient_privilege using message='SERVICE_CALLER_PAIR_HELD';end if;

 perform concierge_service_qa_v1.fence();perform concierge_service_qa_v1.auth_live(p_user,p_session,p_exp);
 a:=concierge_service_qa_v1.launch(p_property,p_review);
 select jsonb_build_object('memberId',m.id,'accountId',m.account_id,'role',m.role) into ctx
 from public.concierge_manager_web_identities i join public.concierge_account_members m on m.id=i.member_id and m.account_id=i.account_id
 where i.auth_user_id=p_user and i.status='active' and i.revoked_at is null and m.status='active' and m.account_id=a;
 if ctx is null or not exists(select from concierge_runtime_qa_v2.concierge_authorize_manager_member_action((ctx->>'memberId')::uuid,p_property,p_action) q where q.result='allow')
 then raise insufficient_privilege using message='SERVICE_MANAGER_DENIED';end if;return ctx;
end$$;

 select count(*),min(x::text)::jsonb into n,live from concierge_writer_cap_v1.auth_live_v1(u,s,r) x;
 if n<>1 or live->>'status' is distinct from 'LIVE' or live->>'userId' is distinct from u::text or live->>'sessionId' is distinct from s::text or live->>'requestId' is distinct from r::text then raise exception 'CONTRACT_SCOPE_DENIED';end if;
 select count(*),min(to_jsonb(x)::text)::jsonb into n,ctx from concierge_runtime_qa_v2.concierge_resolve_manager_web_context(u) x;

 or not exists(select from public.concierge_properties z where z.id=p and z.account_id=(ctx->>'account_id')::uuid and z.status='active' and z.city='Buenos Aires')
 or (select count(*) from public.concierge_properties z where z.account_id=(ctx->>'account_id')::uuid and z.status='active')<>1 then raise exception 'CONTRACT_SCOPE_DENIED';end if;
 v:=array['tjdgrzzeuuihhqnzvaox',r::text,u::text,s::text,e::text,ctx->>'account_id',ctx->>'member_id',ctx->>'member_role',p::text,floor(extract(epoch from clock_timestamp())*1000)::bigint::text];

end$$;
